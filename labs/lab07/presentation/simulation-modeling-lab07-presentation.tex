\section{Параметры и начальные
условия}\label{ux43fux430ux440ux430ux43cux435ux442ux440ux44b-ux438-ux43dux430ux447ux430ux43bux44cux43dux44bux435-ux443ux441ux43bux43eux432ux438ux44f}

\begin{frame}{Параметры и начальные условия}
N = 5000.0 I0 = 150.0 R0 = 0.0 S0 = N - I0 - R0

a = 0.001 \# коэффициент заболеваемости (alpha) b = 0.05 \# коэффициент
выздоровления (beta)

u0 = {[}S0, I0, R0{]} tspan = (0.0, 200.0) p = (a, b)
\end{frame}

\section{Система дифференциальных уравнений для случая
(a)}\label{ux441ux438ux441ux442ux435ux43cux430-ux434ux438ux444ux444ux435ux440ux435ux43dux446ux438ux430ux43bux44cux43dux44bux445-ux443ux440ux430ux432ux43dux435ux43dux438ux439-ux434ux43bux44f-ux441ux43bux443ux447ux430ux44f-a}

\begin{frame}{Система дифференциальных уравнений для случая (a)}
function epidemic\_case\_a!(du, u, p, t) S, I, R = u alpha, beta = p
du{[}1{]} = 0.0 \# dS/dt du{[}2{]} = -beta * I \# dI/dt du{[}3{]} = beta
* I \# dR/dt end

prob\_a = ODEProblem(epidemic\_case\_a!, u0, tspan, p) sol\_a =
solve(prob\_a, Tsit5(), saveat=0.5)
\end{frame}

\section{Система дифференциальных уравнений для случая
(б)}\label{ux441ux438ux441ux442ux435ux43cux430-ux434ux438ux444ux444ux435ux440ux435ux43dux446ux438ux430ux43bux44cux43dux44bux445-ux443ux440ux430ux432ux43dux435ux43dux438ux439-ux434ux43bux44f-ux441ux43bux443ux447ux430ux44f-ux431}

\begin{frame}{Система дифференциальных уравнений для случая (б)}
function epidemic\_case\_b!(du, u, p, t) S, I, R = u alpha, beta = p
du{[}1{]} = -alpha * S * I \# dS/dt du{[}2{]} = alpha * S * I - beta * I
\# dI/dt du{[}3{]} = beta * I \# dR/dt end

prob\_b = ODEProblem(epidemic\_case\_b!, u0, tspan, p) sol\_b =
solve(prob\_b, Tsit5(), saveat=0.5)
\end{frame}
